6: function initRealtimePresence() {
7:     if (!currentUser || !supabaseClient) return;
8:     
9:     if (presenceChannel) {
10:         supabaseClient.removeChannel(presenceChannel);
11:     }
12:     
13:     presenceChannel = supabaseClient.channel('global_presence', {
14:         config: {
15:             presence: {
16:                 key: currentUser.id,
17:             },
18:         },
19:     });
20: 
21:     presenceChannel
22:         .on('presence', { event: 'sync' }, () => {
23:             const newState = presenceChannel.presenceState();
24:             onlineUsers = {};
25:             for (const key in newState) {
26:                 if (newState[key].length > 0) {
27:                     onlineUsers[key] = newState[key][0];
28:                 }
29:             }
30:             renderOnlineUsersBadge();
31:             renderDmUsersList();
32:         })
33:         .on('presence', { event: 'join' }, ({ key, newPresences }) => {
34:             onlineUsers[key] = newPresences[0];
35:             renderOnlineUsersBadge();
36:             renderDmUsersList();
37:         })
38:         .on('presence', { event: 'leave' }, ({ key, leftPresences }) => {
39:             delete onlineUsers[key];
40:             renderOnlineUsersBadge();
41:             renderDmUsersList();
42:         })
43:         .on('broadcast', { event: 'typing' }, ({ payload }) => {
44:             if (payload.userId !== currentUser.id) {
45:                 typingUsers[payload.userId] = {
46:                     timestamp: Date.now(),
47:                     name: payload.name,
48:                     channel: payload.channel // 'global' or dm 'userId'
49:                 };
50:                 renderTypingIndicators();
51:             }
52:         })
53:         .subscribe(async (status) => {
54:             if (status === 'SUBSCRIBED') {
55:                 const pForPresence = currentUserProfile || {};
56:                 await presenceChannel.track({
57:                     id: currentUser.id,
58:                     email: currentUser.email,
59:                     display_name: pForPresence.display_name || pForPresence.displayName || currentUser.email.split('@')[0],
60:                     photo_url: pForPresence.photo_url || pForPresence.photoURL || '',
61:                     avatar_gradient: pForPresence.avatar_gradient || pForPresence.avatarGradient || 'emerald',
62:                     is_admin: pForPresence.is_admin || false
63:                 });
64:             }
65:         });
66:         
67:     setInterval(() => {
68:         const now = Date.now();
69:         let changed = false;
70:         for (const uid in typingUsers) {
71:             if (now - typingUsers[uid].timestamp > 3000) {
72:                 delete typingUsers[uid];
73:                 changed = true;
74:             }
75:         }
76:         if (changed) renderTypingIndicators();
77:     }, 1000);
78: }

80: function broadcastTyping(channelContext) {
81:     if (!presenceChannel || !currentUser) return;
82:     const p = currentUserProfile || {};
83:     presenceChannel.send({
84:         type: 'broadcast',
85:         event: 'typing',
86:         payload: {
87:             userId: currentUser.id,
88:             name: p.display_name || p.displayName || currentUser.email.split('@')[0],
89:             channel: channelContext
90:         }
91:     }).catch(err => console.error(err));
92: }

94: function renderTypingIndicators() {
95:     const globalInd = document.getElementById('global-typing-indicator');
96:     const dmInd = document.getElementById('dm-typing-indicator');
97:     
98:     let globalTypers = [];
99:     let dmTypers = [];
100:     
101:     for (const uid in typingUsers) {
102:         const t = typingUsers[uid];
103:         if (t.channel === 'global') globalTypers.push(t.name);
104:         // if they are typing in DM to ME
105:         if (t.channel === currentUser.id && activeDmRecipient && (activeDmRecipient.id === uid || activeDmRecipient.userId === uid)) {
106:             dmTypers.push(t.name);
107:         }
108:     }
109:     
110:     if (globalInd) {
111:         if (globalTypers.length > 0) {
112:             globalInd.innerHTML = `<span class="typing-anim">💬</span> ${escapeHtml(globalTypers.join(', '))} ${globalTypers.length > 1 ? 'are' : 'is'} typing...`;
113:         } else {
114:             globalInd.innerHTML = '';
115:         }
116:     }
117:     
118:     if (dmInd) {
119:         if (dmTypers.length > 0) {
120:             dmInd.style.display = 'flex';
121:             dmInd.innerHTML = `<span class="typing-anim">💬</span> ${escapeHtml(dmTypers.join(', '))} ${dmTypers.length > 1 ? 'are' : 'is'} typing...`;
122:         } else {
123:             dmInd.style.display = 'none';
124:             dmInd.innerHTML = '';
125:         }
126:     }
127: }

129: function renderOnlineUsersBadge() {
130:     const badge = document.getElementById('online-users-badge');
131:     const dropdown = document.getElementById('online-users-dropdown');
132:     if (!badge || !dropdown) return;
133:     
134:     const count = Object.keys(onlineUsers).length;
135:     badge.innerHTML = `🟢 ${count} Online`;
136:     
137:     let html = '';
138:     for (const uid in onlineUsers) {
139:         const u = onlineUsers[uid];
140:         const isMe = currentUser && uid === currentUser.id;
141:         const nameStr = escapeHtml(u.display_name || u.email || 'Member') + (isMe ? ' (You)' : '');
142:         html += `
143:             <div style="display: flex; align-items: center; gap: 8px;">
144:                 <div style="width: 24px; height: 24px; border-radius: 50%; overflow: hidden; background: transparent; flex-shrink: 0;">

146:                 </div>
147:                 <div style="font-size: 0.8rem; font-weight: 600; color: var(--text-main); white-space: nowrap; overflow: hidden; text-overflow: ellipsis; flex: 1;">
148:                     ${nameStr}
149:                 </div>
150:                 <div style="font-size: 0.6rem; color: var(--accent);">🟢</div>
151:             </div>
152:         `;
153:     }
154:     dropdown.innerHTML = html;
155: }
